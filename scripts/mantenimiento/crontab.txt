# =============================================================================
# Calendario de mantenimiento de PasajeYa  —  crontab del usuario "pasajeya"
# -----------------------------------------------------------------------------
# Instalar con:  crontab -e   (pegar el contenido)   ·   listar:  crontab -l
# Todas las tareas: ruta absoluta + salida a log + horario en hora valle.
# =============================================================================

# Si una tarea falla, cron envia el error a este correo:
MAILTO=equipo-pasajeya@utp.edu.pe
# cron NO hereda el PATH del usuario: se declara explicitamente.
PATH=/usr/local/bin:/usr/bin:/bin

# 1) Health check + autorreparacion  —  cada 15 minutos
*/15 * * * *  /opt/pasajeya/scripts/healthcheck.sh

# 2) Backup diario de la BD (Copia 1: local)  —  todos los dias 2:00 a.m.
0 2 * * *     /opt/pasajeya/scripts/backup_bd.sh >> /var/log/pasajeya/backup.log 2>&1

# 3) Espejo a un segundo medio (Copia 2)  —  2:20 a.m., tras el backup
20 2 * * *    rsync -a /backups/pasajeya/ /mnt/disco2/pasajeya/

# 4) Copia fuera del sitio a la nube (Copia 3)  —  2:40 a.m.
40 2 * * *    rclone copy /backups/pasajeya nube:pasajeya-respaldos

# 5) Limpieza condicional de logs (umbral 85%)  —  domingos 3:30 a.m.
30 3 * * 0    /opt/pasajeya/scripts/limpieza.sh >> /var/log/pasajeya/limpieza.log 2>&1

# 6) Restauracion de PRUEBA  —  dia 1 de cada mes, 5:00 a.m.
0 5 1 * *     /opt/pasajeya/scripts/probar_restauracion.sh >> /var/log/pasajeya/restauracion.log 2>&1
